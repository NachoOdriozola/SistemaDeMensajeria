114void interfazContactos_accion (s_aplicacion *aplicacion, s_interfazContactos *interfazContactos, s_recursosComunesContactosSalas *recursosComunesContactosSalas)
115{
116 sfEvent evento;
117
118
119 sfRenderWindow_pollEvent (aplicacion->renderizado, &evento);
120 switch (evento.type)
121 {
122
123 case sfEvtClosed:
124 aplicacion->aplicacionEjecutandose = DETENER_APLICACION;
125 break;
126
127
128 case sfEvtResized:
129 manejarRedimensionamientoVentana (aplicacion, evento);
130 break;
131
132
133 case sfEvtMouseButtonPressed:
134 if (evento.mouseButton.button == sfMouseLeft)
135 {
136 if (manejarClickAgendarContacto (aplicacion->renderizado, interfazContactos->texto.agendarContacto, &(recursosComunesContactosSalas->habilitaciones), &(interfazContactos->habilitaciones)) == EVENTO_MANEJADO) break;
137 if (manejarClickEscribirAgendarContacto (aplicacion->renderizado, interfazContactos->elementos.barraAgendarContacto, &(interfazContactos->habilitaciones)) == EVENTO_MANEJADO) break;
138 if (manejarClickNotificaciones (aplicacion->renderizado, recursosComunesContactosSalas->texto.notificaciones, &(recursosComunesContactosSalas->habilitaciones), &(interfazContactos->habilitaciones)) == EVENTO_MANEJADO) break;
139 if (manejarClickCerrarVentanaEmergente (aplicacion->renderizado, recursosComunesContactosSalas->texto.cerrarVentanaEmergente, &(recursosComunesContactosSalas->habilitaciones), &(interfazContactos->habilitaciones)) == EVENTO_MANEJADO) break;
140 if (manejarClickEscribirMensaje (aplicacion->renderizado, recursosComunesContactosSalas->elementos.barraEscribirMensaje, &(recursosComunesContactosSalas->habilitaciones)) == EVENTO_MANEJADO) break;
141 if (manejarClickSolapaCambiarInterfaz (aplicacion->renderizado, recursosComunesContactosSalas->elementos.solapaCambiarInterfaz, &(aplicacion->usuario)) == EVENTO_MANEJADO) break;
142 if (manejarClickCambiarInterfazConfig (aplicacion->renderizado, recursosComunesContactosSalas->texto.configuraciones, &(aplicacion->usuario)) == EVENTO_MANEJADO) break;
143 }
144 break;
145
146
147 case sfEvtTextEntered:
148 if (evento.text.unicode < 128)
149 {
150 if (manejarEscribirMensaje (recursosComunesContactosSalas, aplicacion->ventana, evento) == EVENTO_MANEJADO) break;
151 if (manejarEscribirAgendarContacto (interfazContactos, evento) == EVENTO_MANEJADO) break;
152 }
153 break;
154
155
156 case sfEvtKeyPressed:
157 if (evento.key.code == sfKeyEnter)
158 {
159 if (manejarEnviarMensaje (recursosComunesContactosSalas, aplicacion) == EVENTO_MANEJADO) break;
160 if (manejarIntentarAgendarContacto (interfazContactos, aplicacion) == EVENTO_MANEJADO) break;
161 }
162 break;
163
164
165 default:
166 break;
167 }

170void interfazContactos_actualizar (s_aplicacion *aplicacion, s_interfazContactos *interfazContactos, s_recursosComunesContactosSalas *recursosComunesContactosSalas)
171{
172 // RECIBIR MENSAJES DE OTROS USUARIOS
173 char *bufferRespuesta;
174
175 bufferRespuesta = malloc (MAX_BUFFER_RESPUESTA);
176 if (!bufferRespuesta)
177 {
178 perror ("ERROR - Sin memoria.\n");
179 return;
180 }
181 if (recibirSolicitud (aplicacion->sock, bufferRespuesta) == RECIBIO_RESPUESTA)
182 {
183 switch (*bufferRespuesta)
184 {
185 case INDICE_RESPUESTA_SOLICITUD_AMISTAD:
186 agregarNotificacion (&(aplicacion->listaNotificaciones), bufferRespuesta, aplicacion->ventana, aplicacion->mensajes.fuentes);
187 break;
188
189 case INDICE_RESPUESTA_MENSAJE:
190 break;
191 }
192 }
193 free (bufferRespuesta);
194
195
196 int largoBufferMensaje;
197
198 largoBufferMensaje = strlen (recursosComunesContactosSalas->bufferEscribirMensaje);
199
200
201 // TEXTO AUXILIAR ESCRIBIR MENSAJE
202 if (largoBufferMensaje == 0)
203 {
204 if (recursosComunesContactosSalas->habilitaciones.escribirMensaje == DESHABILITAR_ESCRIBIR_MENSAJE)
205 sfText_setString (recursosComunesContactosSalas->texto.auxEscribirMensaje, "Escribir mensaje...");
206 else
207 sfText_setString (recursosComunesContactosSalas->texto.auxEscribirMensaje, "");
208 }
209
210
211 // BARRA DE ESCRIBIENDO MENSAJE
212 if (recursosComunesContactosSalas->habilitaciones.escribirMensaje == HABILITAR_ESCRIBIR_MENSAJE)
213 {
214 if (recursosComunesContactosSalas->habilitaciones.tempPuntoInsercion == 22)
215 {
216 recursosComunesContactosSalas->habilitaciones.tempPuntoInsercion = 0;
217 if (recursosComunesContactosSalas->habilitaciones.puntoInsercion == DESHABILITAR_PUNTO_INSERCION)
218 recursosComunesContactosSalas->habilitaciones.puntoInsercion = HABILITAR_PUNTO_INSERCION;
219 else
220 recursosComunesContactosSalas->habilitaciones.puntoInsercion = DESHABILITAR_PUNTO_INSERCION;
221 }
222 recursosComunesContactosSalas->habilitaciones.tempPuntoInsercion ++;
223 }
224 else
225 {
226 recursosComunesContactosSalas->habilitaciones.puntoInsercion = DESHABILITAR_PUNTO_INSERCION;
227 recursosComunesContactosSalas->habilitaciones.tempPuntoInsercion = 0;
228 }
229
230
231 // TEXTO AUXILIAR AGENDAR CONTACTOS
232 if ((interfazContactos->habilitaciones.escribirAgendarContacto == DESHABILITAR_ESCRIBIR_AGENDAR_CONTACTO) && (strlen (interfazContactos->bufferAgendarContacto) == 0))
233 sfText_setString (interfazContactos->texto.auxAgendarContacto, "Escribir nombre...");
234}

330void intentarSolicitudAmistad (s_aplicacion *aplicacion, s_interfazContactos *interfazContactos)
331{
332 char *bufferSolicitud, *bufferRespuesta;
333 char estadoSolicitud;
334
335 bufferSolicitud = malloc (MAX_BUFFER_SOLICITUD);
336 if (!bufferSolicitud)
337 {
338 perror ("ERROR - Sin memoria.\n");
339 return;
340 }
341 bufferRespuesta = malloc (MAX_BUFFER_RESPUESTA);
342 if (!bufferRespuesta)
343 {
344 perror ("ERROR - Sin memoria.\n");
345 return;
346 }
347
348 sprintf (bufferSolicitud, "%c|%s|%s", INDICE_AGENDARCONTACTO, aplicacion->usuario.nombre, interfazContactos->bufferAgendarContacto);
349 printf ("Enviado: %s\n", bufferSolicitud);
350 enviarYRecibirSolicitud (aplicacion->sock, bufferSolicitud, bufferRespuesta);
351 sscanf (bufferRespuesta, "%c", &estadoSolicitud);
352
353 free (bufferSolicitud);
354 free (bufferRespuesta);
355
356 if (estadoSolicitud == INDICE_RESPUESTA_SOLICITUD_ACEPTADA)
357 printf ("Le envio la solicitud de amistad.\n");
358 else if (estadoSolicitud == INDICE_RESPUESTA_SOLICITUD_RECHAZADA)
359 printf ("No se envio la solicitud de amistad.\n");
360}
