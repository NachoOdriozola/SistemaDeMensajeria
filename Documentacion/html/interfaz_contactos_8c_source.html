129void interfazContactos_accion (s_aplicacion *aplicacion, s_interfazContactos *interfazContactos, s_recursosComunesContactosSalas *recursosComunesContactosSalas)
130{
131 sfEvent evento;
132
133
134 sfRenderWindow_pollEvent (aplicacion->renderizado, &evento);
135 switch (evento.type)
136 {
137
138 case sfEvtClosed:
139 aplicacion->aplicacionEjecutandose = DETENER_APLICACION;
140 break;
141
142
143 case sfEvtResized:
144 if (manejarRedimensionamientoVentana (aplicacion, evento) == ERROR_SIN_MEMORIA)
145 aplicacion->aplicacionEjecutandose = DETENER_APLICACION;
146 break;
147
148
149 case sfEvtMouseButtonPressed:
150 if (evento.mouseButton.button == sfMouseLeft)
151 {
152 if (manejarClickEscribirMensaje (aplicacion->renderizado, recursosComunesContactosSalas, interfazContactos, &(aplicacion->ventana)) == EVENTO_MANEJADO) break;
153 if (manejarClickEscribirAgendarContacto (aplicacion->renderizado, recursosComunesContactosSalas, interfazContactos, &(aplicacion->ventana)) == EVENTO_MANEJADO) break;
154 if (manejarClickNotificaciones (aplicacion, recursosComunesContactosSalas, interfazContactos) == EVENTO_MANEJADO) break;
155 if (manejarClickAgendarContacto (aplicacion, recursosComunesContactosSalas, interfazContactos) == EVENTO_MANEJADO) break;
156 if (manejarClickCerrarVentanaEmergente (aplicacion->renderizado, recursosComunesContactosSalas, interfazContactos) == EVENTO_MANEJADO) break;
157 if (manejarClickSolapaCambiarInterfaz (aplicacion, recursosComunesContactosSalas, interfazContactos) == EVENTO_MANEJADO) break;
158 if (manejarClickCambiarInterfazConfig (aplicacion, recursosComunesContactosSalas, interfazContactos) == EVENTO_MANEJADO) break;
159 }
160 break;
161
162
163 case sfEvtTextEntered:
164 if (evento.text.unicode < 128)
165 {
166 if (manejarEscribirMensaje (recursosComunesContactosSalas, &(aplicacion->ventana), evento) == EVENTO_MANEJADO) break;
167 if (manejarEscribirAgendarContacto (recursosComunesContactosSalas, interfazContactos, &(aplicacion->ventana), evento) == EVENTO_MANEJADO) break;
168 }
169 break;
170
171
172 case sfEvtKeyPressed:
173 if (evento.key.code == sfKeyEnter)
174 {
175 if (manejarEnterEnviarMensaje (recursosComunesContactosSalas, aplicacion) == EVENTO_MANEJADO) break;
176 if (manejarEnterIntentarAgendarContacto (interfazContactos, aplicacion) == EVENTO_MANEJADO) break;
177 }
178 break;
179
180
181 default:
182 break;
183 }

186void interfazContactos_actualizar (s_aplicacion *aplicacion, s_interfazContactos *interfazContactos, s_recursosComunesContactosSalas *recursosComunesContactosSalas)
187{
188 // RECIBIR MENSAJES DE OTROS USUARIOS
189 char *bufferRespuesta;
190
191 bufferRespuesta = malloc (MAX_BUFFER_RESPUESTA);
192 if (!bufferRespuesta)
193 {
194 perror ("ERROR - Sin memoria.\n");
195 return;
196 }
197 if (recibirRespuesta (aplicacion->sock, bufferRespuesta) == RECIBIO_RESPUESTA)
198 {
199 switch (*bufferRespuesta)
200 {
201 case INDICE_RESPUESTA_SOLICITUD_AMISTAD:
202 agregarNotificacion (&(aplicacion->listaNotificaciones), bufferRespuesta, aplicacion->ventana, aplicacion->mensajes.fuentes);
203 break;
204
205 case INDICE_RESPUESTA_MENSAJE:
206 break;
207 }
208 }
209 free (bufferRespuesta);
210
211
212 // --------------- PUNTO DE INSERCION ---------------
213
214 if ((recursosComunesContactosSalas->habilitaciones.escribirMensaje == HABILITAR_ESCRIBIR_MENSAJE) ||
215 (interfazContactos->habilitaciones.escribirAgendarContacto == HABILITAR_ESCRIBIR_AGENDAR_CONTACTO))
216 actualizarPuntoInsercion (&(recursosComunesContactosSalas->habilitaciones.puntoInsercion), &(recursosComunesContactosSalas->habilitaciones.contadorPuntoInsercion));
217 else
218 reiniciarPuntoInsercion (&(recursosComunesContactosSalas->habilitaciones.puntoInsercion), &(recursosComunesContactosSalas->habilitaciones.contadorPuntoInsercion));
219}
