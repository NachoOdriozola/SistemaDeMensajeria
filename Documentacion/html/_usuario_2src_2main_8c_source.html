136 // --------------- INICIALIZAR WINSOCK API ---------------

138 WSADATA wsaData;
139 int resultado;
140
141 resultado = WSAStartup (MAKEWORD (2, 2), &wsaData);
142 if (resultado != 0)

144 printf ("\nERROR - Inicializar Winsock: %d.\n", resultado);

147
148
149 // --------------- INICIALIZAR RENDERIZADO ---------------
150
151 aplicacion->renderizado = sfRenderWindow_create ((sfVideoMode){560, 660}, "Aplicacion", sfDefaultStyle, NULL);

153 {
154 perror ("\nERROR - Crear renderizado.\n");
155 WSACleanup ();

158
159
160 // --------------- INICIALIZAR SOCKET DE LA APLICACION ---------------
161
162 struct sockaddr_in dirCliente;
163 u_long modoSocket = 0; // Establecer socket en modo NO bloqueante.
164
165 aplicacion->sock = socket (AF_INET, SOCK_STREAM, IPPROTO_TCP);
166 if (aplicacion->sock == INVALID_SOCKET)
167 {
168 printf ("\nERROR - Crear el socket: %d.\n", WSAGetLastError ());
169 sfRenderWindow_destroy (aplicacion->renderizado);
170 WSACleanup ();

173
174 dirCliente.sin_family = AF_INET;
175 dirCliente.sin_port = htons (PUERTO); // Escuchar en el puerto asignado.
176 dirCliente.sin_addr.s_addr = inet_addr ("127.0.0.1"); // Aceptar conexiones de cualquier direccion IP.
177 if (connect (aplicacion->sock, (struct sockaddr*)&(dirCliente), sizeof (dirCliente)) == SOCKET_ERROR)
178 {
179 printf ("\nERROR - Conectarse con el servidor: %d.\n", WSAGetLastError ());
180 closesocket (aplicacion->sock);
181 sfRenderWindow_destroy (aplicacion->renderizado);
182 WSACleanup ();
183 return ERROR_INICIALIZACION;
184 }
185 ioctlsocket (aplicacion->sock, FIONBIO, &modoSocket);
186 printf ("-CONECTADO CON EL SERVIDOR-\t");
187
188
189 // --------------- INICIALIZAR RECURSOS GRAFICOS COMUNES ---------------
190
191 if (recursosComunesContactosSalas_inicializar (&(interfaces->recursosComunesContactosSalas)) == ERROR_INICIALIZACION)
192 {
193 closesocket (aplicacion->sock);
194 sfRenderWindow_destroy (aplicacion->renderizado);
195 WSACleanup ();
196 return ERROR_INICIALIZACION;
197 }
198
199
200 // --------------- INICIALIZAR INTERFACES GRAFICAS ESENCIALES ---------------

202 // Interfaz grafica de contactos
203
204 if (interfazContactos_inicializar (&(interfaces->contactos)) == ERROR_INICIALIZACION)

206 recursosComunesContactosSalas_liberar (&(interfaces->recursosComunesContactosSalas));
207 closesocket (aplicacion->sock);
208 sfRenderWindow_destroy (aplicacion->renderizado);
209 WSACleanup ();
210 return ERROR_INICIALIZACION;
211 }

213
214 // Interfaz grafica de salas
215
216 if (interfazSalas_inicializar (&(interfaces->salas)) == ERROR_INICIALIZACION)
217 {
218 interfazContactos_liberar (&(interfaces->contactos));
219 recursosComunesContactosSalas_liberar (&(interfaces->recursosComunesContactosSalas));
220 closesocket (aplicacion->sock);
221 sfRenderWindow_destroy (aplicacion->renderizado);
222 WSACleanup ();
223 return ERROR_INICIALIZACION;

227 // --------------- INICIALIZAR FUENTES ---------------

229 if (inicializarFuentes (&(aplicacion->mensajes.fuentes)) == ERROR_INICIALIZACION)
230 {
231 interfazSalas_liberar (&(interfaces->salas));
232 interfazContactos_liberar (&(interfaces->contactos));
233 recursosComunesContactosSalas_liberar (&(interfaces->recursosComunesContactosSalas));
234 closesocket (aplicacion->sock);
235 sfRenderWindow_destroy (aplicacion->renderizado);
236 WSACleanup ();
237 return ERROR_INICIALIZACION;

239
240
241 // --------------- CREAR E INICIALIZAR LISTA DE MENSAJES ---------------
242
243 unsigned short int i;
244 sfText *mensajes [MAX_MENSAJES_MEMORIA];
245
246 crearListaCircular (&(aplicacion->mensajes.listaMensajes));
247 for (i = 0; i < MAX_MENSAJES_MEMORIA; i ++)
248 {
249 mensajes[i] = sfText_create ();
250 if (!mensajes[i])
251 {
252 printf ("\nERROR - No se pudo crear mensaje[%u].\n", i);
253 vaciarListaCircularConAccion (&(aplicacion->mensajes.listaMensajes), liberarMensaje);
254 liberarFuentes (&(aplicacion->mensajes.fuentes));
255 interfazSalas_liberar (&(interfaces->salas));
256 interfazContactos_liberar (&(interfaces->contactos));
257 recursosComunesContactosSalas_liberar (&(interfaces->recursosComunesContactosSalas));
258 closesocket (aplicacion->sock);
259 sfRenderWindow_destroy (aplicacion->renderizado);
260 WSACleanup ();
261 continue;
262 }
263
264 insertarSegundoCircular (&(aplicacion->mensajes.listaMensajes), &(mensajes[i]), sizeof (sfText*));
265 }
266
267
268 // --------------- CREAR LISTA DE NOTIFICACIONES ---------------
269
270 crearListaSimple (&(aplicacion->listaNotificaciones));
271
272
273 printf ("-INICIALIZACION EXITOSA-\n");
274 return EXITO;
275}
