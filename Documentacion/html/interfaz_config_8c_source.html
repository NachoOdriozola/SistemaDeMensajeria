85void interfazConfig_accion (s_aplicacion *aplicacion, const s_interfazConfig *interfazConfig)
86{
87 sfEvent evento;
88
89 sfRenderWindow_pollEvent (aplicacion->renderizado, &evento);
90 switch (evento.type)
91 {
92
93 case sfEvtClosed:
94 aplicacion->aplicacionEjecutandose = DETENER_APLICACION;
95 break;
96
97
98 case sfEvtResized:
99 if (manejarRedimensionamientoVentana (aplicacion, evento) == ERROR_SIN_MEMORIA)
100 aplicacion->aplicacionEjecutandose = DETENER_APLICACION;
101 break;
102
103
104 case sfEvtMouseButtonPressed:
105 if (evento.mouseButton.button == sfMouseLeft)
106 {
107 if (clickEnRectangulo (aplicacion->renderizado, interfazConfig->elementos.rectanguloVolver))
108 aplicacion->usuario.interfazActual = aplicacion->usuario.ultimaInterfazUtilizada;
109 }
110
111
112 default:
113 break;
114 }
115}
