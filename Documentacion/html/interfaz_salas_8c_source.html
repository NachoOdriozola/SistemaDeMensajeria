105void interfazSalas_accion (s_aplicacion *aplicacion, s_interfazSalas *interfazSalas, s_recursosComunesContactosSalas *recursosComunesContactosSalas)
106{
107 sfEvent evento;
108
109
110 sfRenderWindow_pollEvent (aplicacion->renderizado, &evento);
111 switch (evento.type)
112 {
113
114 case sfEvtClosed:
115 aplicacion->aplicacionEjecutandose = DETENER_APLICACION;
116 break;
117
118
119 case sfEvtResized:
120 manejarRedimensionamientoVentana (aplicacion, evento);
121 break;
122
123
124 case sfEvtMouseButtonPressed:
125 if (evento.mouseButton.button == sfMouseLeft)
126 {
127 if (manejarClickEscribirMensaje (aplicacion->renderizado, recursosComunesContactosSalas->elementos.barraEscribirMensaje, &(recursosComunesContactosSalas->habilitaciones)) == EVENTO_MANEJADO) break;
128 if (manejarClickNotificaciones (aplicacion->renderizado, recursosComunesContactosSalas->texto.notificaciones, &(recursosComunesContactosSalas->habilitaciones)) == EVENTO_MANEJADO) break;
129 if (manejarClickCerrarVentanaEmergente (aplicacion->renderizado, recursosComunesContactosSalas->texto.cerrarVentanaEmergente, &(recursosComunesContactosSalas->habilitaciones)) == EVENTO_MANEJADO) break;
130 if (manejarClickSolapaCambiarInterfaz (aplicacion->renderizado, recursosComunesContactosSalas->elementos.solapaCambiarInterfaz, &(aplicacion->usuario)) == EVENTO_MANEJADO) break;
131 if (manejarClickCambiarInterfazConfig (aplicacion->renderizado, recursosComunesContactosSalas->texto.configuraciones, &(aplicacion->usuario)) == EVENTO_MANEJADO) break;
132 }
133 break;
134
135
136 case sfEvtTextEntered:
137 if (evento.text.unicode < 128)
138 {
139 if (manejarEscribirMensaje (recursosComunesContactosSalas, aplicacion->ventana, evento) == EVENTO_MANEJADO) break;
140 }
141 break;
142
143
144 case sfEvtKeyPressed:
145 if (evento.key.code == sfKeyEnter)
146 {
147 if (manejarEnviarMensaje (recursosComunesContactosSalas, aplicacion) == EVENTO_MANEJADO) break;
148 }
149 break;
150
151
152 default:
153 break;
154 }
155}

157void interfazSalas_actualizar (s_aplicacion *aplicacion, s_interfazSalas *interfazSalas, s_recursosComunesContactosSalas *recursosComunesContactosSalas)
158{
159 int largoBufferMensaje;
160
161 largoBufferMensaje = strlen (recursosComunesContactosSalas->bufferEscribirMensaje);
162
163
164
165 // RECIBIR MENSAJES DE OTROS USUARIOS
166 char *bufferRespuesta;
167
168 bufferRespuesta = malloc (MAX_BUFFER_RESPUESTA);
169 if (!bufferRespuesta)
170 {
171 perror ("ERROR - Sin memoria.\n");
172 return;
173 }
174 if (recibirSolicitud (aplicacion->sock, bufferRespuesta) == RECIBIO_RESPUESTA)
175 {
176 switch (*bufferRespuesta)
177 {
178 case INDICE_RESPUESTA_SOLICITUD_AMISTAD:
179 agregarNotificacion (&(aplicacion->listaNotificaciones), bufferRespuesta, aplicacion->ventana, aplicacion->mensajes.fuentes);
180 break;
181
182 case INDICE_RESPUESTA_MENSAJE:
183 break;
184 }
185 }
186 free (bufferRespuesta);
187
188
189
190 // TEXTO AUXILIAR ESCRIBIR MENSAJE
191 if (largoBufferMensaje == 0)
192 {
193 if (recursosComunesContactosSalas->habilitaciones.escribirMensaje == DESHABILITAR_ESCRIBIR_MENSAJE)
194 sfText_setString (recursosComunesContactosSalas->texto.auxEscribirMensaje, "Escribir mensaje...");
195 else
196 sfText_setString (recursosComunesContactosSalas->texto.auxEscribirMensaje, "");
197 }
198
199
200 // BARRA DE ESCRIBIENDO MENSAJE
201 if (recursosComunesContactosSalas->habilitaciones.escribirMensaje == HABILITAR_ESCRIBIR_MENSAJE)
202 {
203 if (recursosComunesContactosSalas->habilitaciones.tempPuntoInsercion == 22)
204 {
205 recursosComunesContactosSalas->habilitaciones.tempPuntoInsercion = 0;
206 if (recursosComunesContactosSalas->habilitaciones.puntoInsercion == DESHABILITAR_PUNTO_INSERCION)
207 recursosComunesContactosSalas->habilitaciones.puntoInsercion = HABILITAR_PUNTO_INSERCION;
208 else
209 recursosComunesContactosSalas->habilitaciones.puntoInsercion = DESHABILITAR_PUNTO_INSERCION;
210 }
211 recursosComunesContactosSalas->habilitaciones.tempPuntoInsercion ++;
212 }
213 else
214 {
215 recursosComunesContactosSalas->habilitaciones.puntoInsercion = DESHABILITAR_PUNTO_INSERCION;
216 recursosComunesContactosSalas->habilitaciones.tempPuntoInsercion = 0;
217 }
