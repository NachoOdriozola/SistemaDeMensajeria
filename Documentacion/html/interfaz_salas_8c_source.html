128void interfazSalas_accion (t_aplicacion *aplicacion, t_interfazSalas *interfazSalas, t_recursosComunesContactosSalas *recursosComunesContactosSalas)
129{
130 sfEvent evento;
131
132
133 sfRenderWindow_pollEvent (aplicacion->renderizado, &evento);
134 switch (evento.type)
135 {
136
137 case sfEvtClosed:
138 sfRenderWindow_close (aplicacion->renderizado);
139 break;
140
141
142 case sfEvtResized:
143 manejarRedimensionamientoVentanaContactosSalas (aplicacion, recursosComunesContactosSalas, evento);
144 break;
145
146
147 case sfEvtMouseButtonPressed:
148 if (evento.mouseButton.button == sfMouseLeft)
149 {
150 if (manejarClickEscribirMensaje (aplicacion->renderizado, recursosComunesContactosSalas) == EVENTO_MANEJADO) break;
151 if (manejarClickNotificaciones (aplicacion, recursosComunesContactosSalas) == EVENTO_MANEJADO) break;
152 if (manejarClickCerrarVentanaEmergente (aplicacion->renderizado, recursosComunesContactosSalas) == EVENTO_MANEJADO) break;
153 if (manejarClickSolapaCambiarInterfaz (aplicacion, recursosComunesContactosSalas, interfazSalas) == EVENTO_MANEJADO) break;
154 if (manejarClickAreaMensajes (aplicacion->renderizado, recursosComunesContactosSalas) == EVENTO_MANEJADO) break;
155 if (manejarClickCambiarInterfazConfig (aplicacion, recursosComunesContactosSalas, interfazSalas) == EVENTO_MANEJADO) break;
156 }
157 break;
158
159
160 case sfEvtTextEntered:
161 if (evento.text.unicode < 128)
162 {
163 if (manejarEscribirMensaje (recursosComunesContactosSalas, evento) == EVENTO_MANEJADO) break;
164 }
165 break;
166
167
168 case sfEvtKeyPressed:
169 if (evento.key.code == sfKeyEnter)
170 {
171 if (manejarEnterEnviarMensaje (recursosComunesContactosSalas, aplicacion) == EVENTO_MANEJADO) break;
172 }
173 break;
174
175
176 case sfEvtMouseWheelScrolled:
177 if (manejarScrollAreaMensajes (recursosComunesContactosSalas, evento) == EVENTO_MANEJADO) break;
178 break;
179
180
181 default:
182 break;
183 }

186void interfazSalas_actualizar (t_aplicacion *aplicacion, t_interfazSalas *interfazSalas, t_recursosComunesContactosSalas *recursosComunesContactosSalas)
187{
188 // RECIBIR MENSAJES DE OTROS USUARIOS
189 char *bufferRespuesta;
190
191 bufferRespuesta = malloc (MAX_BUFFER_RESPUESTA);
192 if (!bufferRespuesta)
193 {
194 perror ("ERROR - Sin memoria.\n");
195 return;
196 }
197 if (recibirRespuesta (aplicacion->sock, bufferRespuesta) == RECIBIO_RESPUESTA)
198 {
199 switch (*bufferRespuesta)
200 {
201 case INDICE_RESPUESTA_AGENDAR_CONTACTO:
202 agregarNotificacion (&(aplicacion->listaNotificaciones), bufferRespuesta, aplicacion->mensajes.fuentes);
203 break;
204
205 case INDICE_RESPUESTA_MENSAJE:
206 break;
207 }
208 }
209 free (bufferRespuesta);
210
211
212 // --------------- PUNTO DE INSERCION ---------------
213
214 if (recursosComunesContactosSalas->habilitaciones.escribirMensaje == HABILITAR_ESCRIBIR_MENSAJE)
215 actualizarPuntoInsercion (&(recursosComunesContactosSalas->habilitaciones.puntoInsercion), &(recursosComunesContactosSalas->habilitaciones.contadorPuntoInsercion));
216 else
217 reiniciarPuntoInsercion (&(recursosComunesContactosSalas->habilitaciones.puntoInsercion), &(recursosComunesContactosSalas->habilitaciones.contadorPuntoInsercion));
