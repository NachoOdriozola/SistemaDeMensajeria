118void interfazSalas_accion (s_aplicacion *aplicacion, s_interfazSalas *interfazSalas, s_recursosComunesContactosSalas *recursosComunesContactosSalas)
119{
120 sfEvent evento;
121
122
123 sfRenderWindow_pollEvent (aplicacion->renderizado, &evento);
124 switch (evento.type)
125 {
126
127 case sfEvtClosed:
128 aplicacion->aplicacionEjecutandose = DETENER_APLICACION;
129 break;
130
131
132 case sfEvtResized:
133 if (manejarRedimensionamientoVentana (aplicacion, evento) == ERROR_SIN_MEMORIA)
134 aplicacion->aplicacionEjecutandose = DETENER_APLICACION;
135 break;
136
137
138 case sfEvtMouseButtonPressed:
139 if (evento.mouseButton.button == sfMouseLeft)
140 {
141 if (manejarClickEscribirMensaje (aplicacion->renderizado, recursosComunesContactosSalas, &(aplicacion->ventana)) == EVENTO_MANEJADO) break;
142 if (manejarClickNotificaciones (aplicacion, recursosComunesContactosSalas) == EVENTO_MANEJADO) break;
143 if (manejarClickCerrarVentanaEmergente (aplicacion->renderizado, recursosComunesContactosSalas) == EVENTO_MANEJADO) break;
144 if (manejarClickSolapaCambiarInterfaz (aplicacion, recursosComunesContactosSalas, interfazSalas) == EVENTO_MANEJADO) break;
145 if (manejarClickCambiarInterfazConfig (aplicacion, recursosComunesContactosSalas, interfazSalas) == EVENTO_MANEJADO) break;
146 }
147 break;
148
149
150 case sfEvtTextEntered:
151 if (evento.text.unicode < 128)
152 {
153 if (manejarEscribirMensaje (recursosComunesContactosSalas, &(aplicacion->ventana), evento) == EVENTO_MANEJADO) break;
154 }
155 break;
156
157
158 case sfEvtKeyPressed:
159 if (evento.key.code == sfKeyEnter)
160 {
161 if (manejarEnterEnviarMensaje (recursosComunesContactosSalas, aplicacion) == EVENTO_MANEJADO) break;
162 }
163 break;
164
165
166 default:
167 break;
168 }
169}

171void interfazSalas_actualizar (s_aplicacion *aplicacion, s_interfazSalas *interfazSalas, s_recursosComunesContactosSalas *recursosComunesContactosSalas)
172{
173 // RECIBIR MENSAJES DE OTROS USUARIOS
174 char *bufferRespuesta;
175
176 bufferRespuesta = malloc (MAX_BUFFER_RESPUESTA);
177 if (!bufferRespuesta)
178 {
179 perror ("ERROR - Sin memoria.\n");
180 return;
181 }
182 if (recibirRespuesta (aplicacion->sock, bufferRespuesta) == RECIBIO_RESPUESTA)
183 {
184 switch (*bufferRespuesta)
185 {
186 case INDICE_RESPUESTA_SOLICITUD_AMISTAD:
187 agregarNotificacion (&(aplicacion->listaNotificaciones), bufferRespuesta, aplicacion->ventana, aplicacion->mensajes.fuentes);
188 break;
189
190 case INDICE_RESPUESTA_MENSAJE:
191 break;
192 }
193 }
194 free (bufferRespuesta);
195
196
197 // --------------- PUNTO DE INSERCION ---------------
198
199 if (recursosComunesContactosSalas->habilitaciones.escribirMensaje == HABILITAR_ESCRIBIR_MENSAJE)
200 actualizarPuntoInsercion (&(recursosComunesContactosSalas->habilitaciones.puntoInsercion), &(recursosComunesContactosSalas->habilitaciones.contadorPuntoInsercion));
201 else
202 reiniciarPuntoInsercion (&(recursosComunesContactosSalas->habilitaciones.puntoInsercion), &(recursosComunesContactosSalas->habilitaciones.contadorPuntoInsercion));
