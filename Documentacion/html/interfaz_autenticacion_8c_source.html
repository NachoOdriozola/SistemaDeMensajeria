131void interfazAutenticacion_accion (t_aplicacion *aplicacion, const t_interfazAutenticacion *interfazAutenticacion, t_recursosComunesAutenticacionRegistro *recursosComunesAutenticacionRegistro)
132{
133 sfEvent evento;
134
135
136 sfRenderWindow_pollEvent (aplicacion->renderizado, &evento);
137 switch (evento.type)
138 {
139
140 case sfEvtClosed:
141 sfRenderWindow_close (aplicacion->renderizado);
142 break;
143
144
145 case sfEvtResized:
146 sfRenderWindow_setSize (aplicacion->renderizado, (sfVector2u){560, 660});
147 break;
148
149
150 case sfEvtMouseButtonPressed:
151 if (evento.mouseButton.button == sfMouseLeft)
152 {
153 if (manejarClickEscribirNombre (aplicacion->renderizado, recursosComunesAutenticacionRegistro) == EVENTO_MANEJADO) break;
154 if (manejarClickEscribirContrasenia (aplicacion->renderizado, recursosComunesAutenticacionRegistro) == EVENTO_MANEJADO) break;
155 if (manejarClickIntentarAutenticacion (aplicacion, recursosComunesAutenticacionRegistro) == EVENTO_MANEJADO) break;
156 if (manejarClickCambiarInterfazRegistro (aplicacion, recursosComunesAutenticacionRegistro, interfazAutenticacion) == EVENTO_MANEJADO) break;
157 if (manejarClickGuardarAutenticacion (aplicacion->renderizado, recursosComunesAutenticacionRegistro) == EVENTO_MANEJADO) break;
158 }
159 break;
160
161
162 case sfEvtTextEntered:
163 if (evento.text.unicode < 128)
164 {
165 if (manejarEscribirNombre (recursosComunesAutenticacionRegistro, evento) == EVENTO_MANEJADO) break;
166 if (manejarEscribirContrasenia (recursosComunesAutenticacionRegistro, evento) == EVENTO_MANEJADO) break;
167 }
168 break;
169
170
171 case sfEvtKeyPressed:
172 if (evento.key.code == sfKeyEnter)
173 {
174 if (manejarEnterIntentarAutenticacion (aplicacion, recursosComunesAutenticacionRegistro) == EVENTO_MANEJADO) break;
175 }
176 break;
177
178
179 default:
180 break;
181 }
182}

184void interfazAutenticacion_actualizar (t_interfazAutenticacion *interfazAutenticacion, t_recursosComunesAutenticacionRegistro *recursosComunesAutenticacionRegistro)

186 int largoBufferEscribirNombre, largoBufferEscribirContrasenia;

189 largoBufferEscribirContrasenia = strlen (recursosComunesAutenticacionRegistro->bufferContrasenia);

270{
271 char bufferSolicitud [MAX_BUFFER_SOLICITUD], bufferRespuesta [MAX_BUFFER_RESPUESTA];
272 char estadoRespuesta;
273 int id;
274
275 snprintf (bufferSolicitud, MAX_BUFFER_SOLICITUD, "%c|%s|%s", INDICE_SOLICITUD_AUTENTICACION, recursosComunesAutenticacionRegistro->bufferNombre, recursosComunesAutenticacionRegistro->bufferContrasenia);
276 enviarSolicitudYRecibirRespuesta (aplicacion->sock, bufferSolicitud, bufferRespuesta);
277 sscanf (bufferRespuesta, "%c|%d", &estadoRespuesta, &id);

281 aplicacion->usuario.id = id;
