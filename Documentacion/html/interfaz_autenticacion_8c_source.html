119void interfazAutenticacion_accion (s_aplicacion *aplicacion, const s_interfazAutenticacion *interfazAutenticacion, s_recursosComunesAutenticacionRegistro *recursosComunesAutenticacionRegistro)
120{
121 sfEvent evento;
122
123
124 sfRenderWindow_pollEvent (aplicacion->renderizado, &evento);
125 switch (evento.type)
126 {
127
128 case sfEvtClosed:
129 aplicacion->aplicacionEjecutandose = DETENER_APLICACION;
130 break;
131
132
133 case sfEvtResized:
134 if (manejarRedimensionamientoVentana (aplicacion, evento) == ERROR_SIN_MEMORIA)
135 aplicacion->aplicacionEjecutandose = DETENER_APLICACION;
136 break;
137
138
139 case sfEvtMouseButtonPressed:
140 if (evento.mouseButton.button == sfMouseLeft)
141 {
142 if (manejarClickEscribirNombre (aplicacion->renderizado, recursosComunesAutenticacionRegistro) == EVENTO_MANEJADO) break;
143 if (manejarClickEscribirContrasenia (aplicacion->renderizado, recursosComunesAutenticacionRegistro) == EVENTO_MANEJADO) break;
144 if (manejarClickIntentarAutenticar (aplicacion, recursosComunesAutenticacionRegistro) == EVENTO_MANEJADO) break;
145 if (manejarClickCambiarInterfazRegistro (aplicacion, recursosComunesAutenticacionRegistro, interfazAutenticacion) == EVENTO_MANEJADO) break;
146 if (manejarClickGuardarAutenticacion (aplicacion->renderizado, recursosComunesAutenticacionRegistro) == EVENTO_MANEJADO) break;
147 }
148 break;
149
150
151 case sfEvtTextEntered:
152 if (evento.text.unicode < 128)
153 {
154 if (manejarEscribirNombre (recursosComunesAutenticacionRegistro, evento) == EVENTO_MANEJADO) break;
155 if (manejarEscribirContrasenia (recursosComunesAutenticacionRegistro, evento) == EVENTO_MANEJADO) break;
156 }
157 break;
158
159
160 case sfEvtKeyPressed:
161 if (evento.key.code == sfKeyEnter)
162 {
163 if (manejarEnterIntentarAutenticacion (aplicacion, recursosComunesAutenticacionRegistro) == EVENTO_MANEJADO) break;
164 }
165 break;
166
167
168 default:
169 break;
170 }
171}

173void interfazAutenticacion_actualizar (s_interfazAutenticacion *interfazAutenticacion, s_recursosComunesAutenticacionRegistro *recursosComunesAutenticacionRegistro)
174{
175 int largoBufferEscribirNombre, largoBufferEscribirContrasenia;
176
177 largoBufferEscribirNombre = strlen (recursosComunesAutenticacionRegistro->bufferNombre);
178 largoBufferEscribirContrasenia = strlen (recursosComunesAutenticacionRegistro->bufferContrasenia);
179
180
181 if ((largoBufferEscribirNombre > 0) && (largoBufferEscribirContrasenia > 0))
182 recursosComunesAutenticacionRegistro->habilitaciones.ingresar = HABILITAR_INGRESAR;
183 else
184 recursosComunesAutenticacionRegistro->habilitaciones.ingresar = DESHABILITAR_INGRESAR;
185
186
187 // --------------- PUNTO DE INSERCION ---------------

246void intentarIngreso (s_aplicacion *aplicacion, s_recursosComunesAutenticacionRegistro *recursosComunesAutenticacionRegistro)
247{
248 char *bufferSolicitud, *bufferRespuesta;
249 char estadoSolicitud;
250 int id;
251
252 bufferSolicitud = malloc (MAX_BUFFER_SOLICITUD);
253 if (!bufferSolicitud)
254 {
255 perror ("ERROR - Sin memoria.\n");
256 return;
257 }
258 bufferRespuesta = malloc (MAX_BUFFER_RESPUESTA);
259 if (!bufferRespuesta)
260 {
261 free (bufferSolicitud);
262 perror ("ERROR - Sin memoria.\n");
263 return;
264 }
265
266 sprintf (bufferSolicitud, "%c|%s|%s", INDICE_AUTENTICACION, recursosComunesAutenticacionRegistro->bufferNombre, recursosComunesAutenticacionRegistro->bufferContrasenia);
267 enviarSolicitudYRecibirRespuesta (aplicacion->sock, bufferSolicitud, bufferRespuesta);
268 sscanf (bufferRespuesta, "%c|%d", &estadoSolicitud, &id);
269
270 free (bufferSolicitud);
271 free (bufferRespuesta);
272
273 if (estadoSolicitud == INDICE_RESPUESTA_SOLICITUD_ACEPTADA)
274 {
275 aplicacion->usuario.interfazActual = INTERFAZ_CONTACTOS;
276 aplicacion->usuario.id = id;
277 strcpy (aplicacion->usuario.nombre, recursosComunesAutenticacionRegistro->bufferNombre);
278 if (recursosComunesAutenticacionRegistro->habilitaciones.guardarAutenticacion == HABILITAR_GUARDAR_AUTENTICACION)
279 guardarDatosEnArchivo (aplicacion->usuario.id, recursosComunesAutenticacionRegistro->bufferNombre);
280 }
281 else
282 printf ("Usuario o contrasenia incorrectos.\n");
283}
