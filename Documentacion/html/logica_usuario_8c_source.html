27void ingresoTexto (char *buffer, int tamMaxBuffer, sfEvent eventoChar)
28{
29 int largoBuffer;
30
31 largoBuffer = strlen (buffer);
32
33 if ((eventoChar.text.unicode != 13) && (eventoChar.text.unicode != 8)) //Detecta que no sea la tecla "Enter" ni "Backspace"
34 {
35 if (largoBuffer < tamMaxBuffer - 1)
36 {
37 buffer [largoBuffer] = (char)eventoChar.text.unicode;
38 buffer [largoBuffer + 1] = '\0';
39 }
40 }
41 else if ((eventoChar.text.unicode == 8) && (largoBuffer > 0)) //Detecta que sea la tecla "Backspace"
42 buffer [largoBuffer - 1] = '\0';

369void eventoMaximizadoAutomatico (sfRenderWindow *renderizado, s_ventana *ventana)
370{
371 sfEvent eventoPrimerMaximizado;
372 sfVector2f nuevoTamVentana;
373
374 // Procesar todos los eventos ocurridos hasta encontrar el de maximizado de la ventana.
375 while (sfRenderWindow_pollEvent (renderizado, &eventoPrimerMaximizado))
376 {
377 if (eventoPrimerMaximizado.type == sfEvtResized)
378 {
379 nuevoTamVentana.x = eventoPrimerMaximizado.size.width;
380 nuevoTamVentana.y = eventoPrimerMaximizado.size.height;
381 break;
382 }
383 }
384 ventana->tamVentana = nuevoTamVentana;
385}
