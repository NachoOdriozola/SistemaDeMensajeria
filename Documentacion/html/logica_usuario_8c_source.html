162void ingresarCaracterABuffer (char *buffer, int tamMaxBuffer, sfEvent eventoChar)
163{
164 int largoBuffer;
165
166 largoBuffer = strlen (buffer);
167
168 if (eventoChar.text.unicode == 13) // Si la tecla es "Enter" retorna.
169 return;
170
171 if (eventoChar.text.unicode != 8) //Si la tecla no es "Backspace".
172 {
173 if (largoBuffer < tamMaxBuffer - 1)
174 {
175 buffer [largoBuffer] = (char)eventoChar.text.unicode;
176 buffer [largoBuffer + 1] = '\0';
177 }
178 }
179 else if (largoBuffer > 0) // Si la tecla es "Backspace".
180 buffer [largoBuffer - 1] = '\0';
181}
