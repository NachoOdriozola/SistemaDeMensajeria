43{

50 if (eventoPrimerMaximizado.type == sfEvtResized)
51 {
52 nuevoTamVentana.x = eventoPrimerMaximizado.size.width;
53 nuevoTamVentana.y = eventoPrimerMaximizado.size.height;
54 break;
55 }
56 }

244void ingresarCaracterABuffer (char *buffer, int tamMaxBuffer, sfEvent eventoChar)
245{
246 int largoBuffer;
247
248 largoBuffer = strlen (buffer);
249
250 if (eventoChar.text.unicode == 13) // Si la tecla es "Enter" retorna.
251 return;
252
253 if (eventoChar.text.unicode != 8) //Si la tecla no es "Backspace".
254 {
255 if (largoBuffer < tamMaxBuffer - 1)
256 {
257 buffer [largoBuffer] = (char)eventoChar.text.unicode;
258 buffer [largoBuffer + 1] = '\0';
259 }
260 }
261 else if (largoBuffer > 0) // Si la tecla es "Backspace".
262 buffer [largoBuffer - 1] = '\0';
